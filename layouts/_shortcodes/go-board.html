{{- $position := .Position -}} {{- if not .IsNamedParams -}} {{- errorf "go-board at %s: use named arguments (src, caption, and optional move or path)" $position -}} {{- else -}} {{- $allowedParams := dict "src" true "caption" true "move" true "path" true -}} {{- $valid := true -}} {{- range $name, $_ := .Params -}} {{- if not (isset $allowedParams $name) -}} {{- $valid = false -}} {{- errorf "go-board at %s: unknown named argument %q" $position $name -}} {{- end -}} {{- end -}} {{- $src := strings.TrimSpace (printf "%v" (.Get "src")) -}} {{- $caption := strings.TrimSpace (printf "%v" (.Get "caption")) -}} {{- $hasMove := isset .Params "move" -}} {{- $hasPath := isset .Params "path" -}} {{- $move := "0" -}} {{- if $hasMove -}} {{- $move = strings.TrimSpace (printf "%v" (.Get "move")) -}} {{- end -}} {{- $path := "" -}} {{- if $hasPath -}} {{- $path = strings.TrimSpace (printf "%v" (.Get "path")) -}} {{- end -}} {{- if eq $src "" -}} {{- $valid = false -}} {{- errorf "go-board at %s: src must be non-empty" $position -}} {{- else if not (strings.HasSuffix (strings.ToLower $src) ".sgf") -}} {{- $valid = false -}} {{- errorf "go-board at %s: src must name a local .sgf resource, got %q" $position $src -}} {{- end -}} {{- if eq $caption "" -}} {{- $valid = false -}} {{- errorf "go-board at %s: caption must be non-empty for %q" $position $src -}} {{- end -}} {{- if and $hasMove $hasPath -}} {{- $valid = false -}} {{- errorf "go-board at %s: move and path are mutually exclusive for %q" $position $src -}} {{- end -}} {{- if and $hasMove (eq (len (findRE `^[0-9]+$` $move)) 0) -}} {{- $valid = false -}} {{- errorf "go-board at %s: move %q must be a non-negative integer" $position $move -}} {{- end -}} {{- if and $hasPath (eq (len (findRE `^(N[0-9]+|B[1-9][0-9]*)+$` $path)) 0) -}} {{- $valid = false -}} {{- errorf "go-board at %s: path %q must be a sequence of N and B tokens" $position $path -}} {{- end -}} {{- $resource := .Page.Resources.Get $src -}} {{- if and (ne $src "") (strings.HasSuffix (strings.ToLower $src) ".sgf") (not $resource) -}} {{- $valid = false -}} {{- errorf "go-board at %s: resource %q not found in page bundle %q" $position $src .Page.RelPermalink -}} {{- end -}} {{- if and $valid $resource -}} {{- $selectorKind := "move" -}} {{- $selectorValue := $move -}} {{- if $hasPath -}} {{- $selectorKind = "path" -}} {{- $selectorValue = $path -}} {{- end -}} {{- $selector := printf "%s:%s" $selectorKind $selectorValue -}} {{- $figureKey := printf "%s|%s|%s|%d" .Page.Permalink $selector $src .Ordinal -}} {{- $figureId := printf "go-board-%s" (substr (md5 $figureKey) 0 12) -}} {{- $captionId := printf "%s-caption" $figureId -}} {{- $controlsId := printf "%s-controls-label" $figureId -}} {{- $variationsId := printf "%s-variations" $figureId -}} {{- $variationsLabelId := printf "%s-variations-label" $figureId -}} {{- $noticeId := printf "%s-local-notice" $figureId -}} {{- $moveTemplate := T "goBoardMove" -}} {{- end -}} {{- end -}}
